cmake_minimum_required(VERSION 3.27)
project(mtplx_ane_ffn_ext LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_SHARED_LIBS "Build extensions as a shared library" ON)
find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT)
find_package(nanobind CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m mlx --cmake-dir OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE MLX_ROOT)
find_package(MLX CONFIG REQUIRED)
add_library(mtplx_ane_client OBJECT ${CMAKE_CURRENT_LIST_DIR}/ane_client.mm)
target_compile_options(mtplx_ane_client PRIVATE -fobjc-arc)
add_library(mtplx_ane_ffn_ext)
target_sources(mtplx_ane_ffn_ext PUBLIC ${CMAKE_CURRENT_LIST_DIR}/ane_ffn.cpp $<TARGET_OBJECTS:mtplx_ane_client>)
target_include_directories(mtplx_ane_ffn_ext PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(mtplx_ane_ffn_ext PUBLIC mlx "-framework Foundation" "-framework IOSurface" "-framework Metal")
nanobind_add_module(_ext NB_STATIC LTO NOMINSIZE NB_DOMAIN mlx ${CMAKE_CURRENT_LIST_DIR}/bindings.cpp)
target_link_libraries(_ext PRIVATE mtplx_ane_ffn_ext)
if(BUILD_SHARED_LIBS)
  target_link_options(_ext PRIVATE -Wl,-rpath,@loader_path)
endif()
